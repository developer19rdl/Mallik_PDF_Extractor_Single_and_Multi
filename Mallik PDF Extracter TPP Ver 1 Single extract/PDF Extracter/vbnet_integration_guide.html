3. Server Executable Deployment & VB.NET Auto-Start Background Process

pdf_upload_server.exe (in dist\) is a standalone binary. Place it alongside your application and run this helper on form load to guarantee it is active:

VB.NET — Automatic Server Process Launcher Helper Sub 
Private Sub EnsurePdfServerRunning()
    Try
        If Process.GetProcessesByName("pdf_upload_server").Length = 0 Then
            Dim exePath As String = Path.Combine(Application.StartupPath, "pdf_upload_server.exe")
            If File.Exists(exePath) Then
                Dim psi As New ProcessStartInfo(exePath) With {.WindowStyle = ProcessWindowStyle.Hidden}
                Process.Start(psi)
            End If
        End If
    Catch ex As Exception
        Console.WriteLine("Server auto-start check error: " & ex.Message)
    End Try
End Sub

Private Sub Button7_Click(sender As Object, e As EventArgs) Handles Button7.Click
    ' 1. Show OpenFileDialog to select PDF file
    Dim openFileDialog As New OpenFileDialog()
    openFileDialog.Filter = "PDF Files (*.pdf)|*.pdf|All Files (*.*)|*.*"
    openFileDialog.Title = "Select a PDF File"

    If openFileDialog.ShowDialog() = DialogResult.OK Then
        Dim selectedFilePath As String = openFileDialog.FileName
        Dim destinationFolder As String = "\\MEIPLSERVER\TestFolder"

        If Not Directory.Exists(destinationFolder) Then
            MessageBox.Show("Destination folder does not exist or is not accessible.")
            Return
        End If

        Dim originalFileName As String = Path.GetFileNameWithoutExtension(selectedFilePath)
        Dim fileExtension As String = Path.GetExtension(selectedFilePath)
        Dim destinationFilePath As String = Path.Combine(destinationFolder, Path.GetFileName(selectedFilePath))

        Dim fileCounter As Integer = 1
        While File.Exists(destinationFilePath)
            destinationFilePath = Path.Combine(destinationFolder, originalFileName & "_" & fileCounter.ToString() & fileExtension)
            fileCounter += 1
        End While

        Try
            File.Copy(selectedFilePath, destinationFilePath, True)
            MessageBox.Show("PDF file saved to: " & destinationFilePath)
            pdfpath = destinationFilePath

            ' 2. Convert PDF to Base64 URI & Upload
            Dim base64String As String = ConvertToBase64(destinationFilePath)
            UploadPdf(base64String)
        Catch ex As Exception
            MessageBox.Show("Error copying or uploading file: " & ex.Message)
        End Try
    End If
End Sub

Private Function ConvertToBase64(filePath As String) As String
    Try
        Dim fileBytes() As Byte = File.ReadAllBytes(filePath)
        Return "data:application/pdf;base64," & Convert.ToBase64String(fileBytes)
    Catch ex As Exception
        MessageBox.Show("Error converting file to Base64: " & ex.Message)
        Return String.Empty
    End Try
End Function

Private Sub UploadPdf(base64String As String)
    If String.IsNullOrEmpty(base64String) Then
        MessageBox.Show("Base64 string is empty.")
        Return
    End If

    Dim url As String = "http://127.0.0.1:8004/upload_pdf"

    Try
        Dim request As HttpWebRequest = CType(WebRequest.Create(url), HttpWebRequest)
        request.Method = "POST"
        request.ContentType = "application/json"

        Dim json As String = "{"""file"":""" & base64String & """"}
        Dim postData As Byte() = Encoding.UTF8.GetBytes(json)
        request.ContentLength = postData.Length

        Using requestStream As Stream = request.GetRequestStream()
            requestStream.Write(postData, 0, postData.Length)
        End Using

        Using response As WebResponse = request.GetResponse()
            Using responseStream As Stream = response.GetResponseStream()
                Using reader As New StreamReader(responseStream)
                    Dim responseFromServer As String = reader.ReadToEnd()
                    Dim jsonResponse As JObject = JObject.Parse(responseFromServer)

                    If jsonResponse.ContainsKey("csv") Then
                        Dim csvBase64 As String = jsonResponse("csv").ToString()
                        Dim fileBytes() As Byte = Convert.FromBase64String(csvBase64)
                        Dim csvData As String = Encoding.UTF8.GetString(fileBytes)

                        ' Parse CSV and update WinForms Controls
                        TextBox21.Text = ExtractWord(csvData, "Machine Name\s*(.*),").Replace(",", "")
                        TextBox16.Text = ExtractWord(csvData, "Material Name\s*(.*)").Replace("E-L2", "").Replace(",", "").Trim()
                        TextBox22.Text = ExtractWord(csvData, "Thickness\s*(.*)").Replace(",", "")
                        TextBox24.Text = ExtractWord(csvData, "Raw Material Weight\s*(.*)").Replace("Process", "").Replace(",", "").Trim()
                        TextBox27.Text = ExtractWord(csvData, "Laser Material Name\s*(.*),").Replace(",", "")
                        TextBox25.Text = ExtractWord(csvData, "Process Length\s*(.*)").Replace("Process", "").Replace(",", "").Trim()
                        TextBox18.Text = ExtractWordmaterialSize(csvData, "Material Size\s*(.*?)$").Replace("Thickness.*", "").Replace(",", "").Trim()
                        TextBox26.Text = ExtractWordProcessQty(csvData, "Process Quantity\s*(\d+(\.\d+)?)")
                    Else
                        MessageBox.Show("CSV data not found in the response.")
                    End If
                End Using
            End Using
        End Using
    Catch ex As Exception
        MessageBox.Show("Error uploading file: " & ex.Message)
    End Try
End Sub

Private Function ExtractWord(data As String, pattern As String) As String
    Dim match As Match = Regex.Match(data, pattern)
    If match.Success Then Return match.Groups(1).Value.Split(" "c)(0)
    Return String.Empty
End Function

Private Function ExtractWordmaterialSize(data As String, pattern As String) As String
    Dim match As Match = Regex.Match(data, pattern, RegexOptions.Multiline)
    If match.Success Then
        Return Regex.Replace(match.Groups(1).Value, "Thickness.*", "").Trim()
    End If
    Return String.Empty
End Function

Private Function ExtractWordProcessQty(data As String, pattern As String) As String
    Dim match As Match = Regex.Match(data, pattern)
    If match.Success Then
        Dim integerValue As Integer
        If Integer.TryParse(match.Groups(1).Value.Split(" "c)(0).Split("."c)(0), integerValue) Then
            Return integerValue.ToString()
        End If
    End If
    Return String.Empty
End Function
